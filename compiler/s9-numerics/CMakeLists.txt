cmake_minimum_required(VERSION 3.20)
project(S9Numerics LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(MLIR 19.1.1 EXACT REQUIRED CONFIG)
list(APPEND CMAKE_MODULE_PATH "${MLIR_CMAKE_DIR}")
include(AddLLVM)
include(AddMLIR)
include_directories(SYSTEM ${LLVM_INCLUDE_DIRS} ${MLIR_INCLUDE_DIRS})
add_definitions(${LLVM_DEFINITIONS})
add_executable(s9-opt s9-opt.cpp)
target_link_libraries(s9-opt PRIVATE MLIROptLib MLIR)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME s9-ir-contracts COMMAND ${Python3_EXECUTABLE}
  ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_ir.py $<TARGET_FILE:s9-opt>)
